```typescript
import { exec } from 'child_process';
import { promisify } from 'util';
import { writeFile, readFile, unlink } from 'fs/promises';
import path from 'path';

const execAsync = promisify(exec);

export class ClaudeService {
  private tempDir = '/tmp/oilfield-ai';

  constructor() {
    // Ensure temp directory exists
    this.ensureTempDir();
  }

  private async ensureTempDir() {
    const { exec } = require('child_process');
    exec(`mkdir -p ${this.tempDir}`);
  }

  /**
   * Generate investment report for a well
   */
  async generateInvestmentReport(well: Well, valuation: Valuation): Promise<string> {
    const prompt = this.buildReportPrompt(well, valuation);

    // Write prompt to temp file
    const promptFile = path.join(this.tempDir, `prompt-${well.id}.txt`);
    await writeFile(promptFile, prompt);

    try {
      // Call Claude CLI
      const { stdout } = await execAsync(
        `claude -p "${promptFile}" --model claude-sonnet-4.5`
      );

      // Clean up
      await unlink(promptFile);

      return stdout;
    } catch (error) {
      console.error('Claude CLI error:', error);
      throw new Error('Failed to generate investment report');
    }
  }
```

### API Routes

```typescript
// src/routes/ai.routes.ts

import { Router } from 'express';
import { ClaudeService } from '../services/claudeService';

const router = Router();
const claudeService = new ClaudeService();

/**
 * POST /api/wells/:id/generate-report
 * Generate AI investment report
 */
router.post('/wells/:id/generate-report', async (req, res) => {
  try {
    const { id } = req.params;

    // Get well and valuation data
    const well = await db('wells').where({ id }).first();
    const valuation = await db('valuations')
      .where({ well_id: id })
      .orderBy('valuation_date', 'desc')
      .first();

    if (!well || !valuation) {
      return res.status(404).json({ error: 'Well or valuation not found' });
    }

    // Generate report (async - returns immediately)
    const reportPromise = claudeService.generateInvestmentReport(well, valuation);

    // For demo: wait for completion (in production, make this async with job queue)
    const report = await reportPromise;

    res.json({
      wellId: well.well_id,
      report,
      generatedAt: new Date().toISOString(),
      format: 'markdown'
    });
  } catch (error) {
    console.error('Error generating report:', error);
    res.status(500).json({ error: 'Failed to generate report' });
  }
});

/**
 * POST /api/chat
 * Conversational well analysis
 */
router.post('/chat', async (req, res) => {
  try {
    const { wellId, message, conversationHistory } = req.body;

    if (!wellId || !message) {
      return res.status(400).json({ error: 'wellId and message are required' });
    }

    const response = await claudeService.chatAboutWell(
      wellId,
      message,
      conversationHistory || []
    );

    res.json({
      wellId,
      response,
      timestamp: new Date().toISOString()
    });
  } catch (error) {
    console.error('Error in chat:', error);
    res.status(500).json({ error: 'Failed to process chat message' });
  }
});

/**
 * GET /api/wells/:id/narrative
 * Get AI-generated well narrative
 */
router.get('/wells/:id/narrative', async (req, res) => {
  try {
    const { id } = req.params;

    // Check if narrative already exists
    let narrative = await db('well_narratives')
      .where({ well_id: id })
      .where('created_at', '>', db.raw("NOW() - INTERVAL '7 days'")) // Cache for 7 days
      .first();

    if (!narrative) {
      // Generate new narrative
      const well = await db('wells').where({ id }).first();
      const text = await claudeService.generateWellNarrative(well);

      // Cache it
      await db('well_narratives').insert({
        well_id: id,
        narrative: text,
        created_at: new Date()
      });

      narrative = { narrative: text };
    }

    res.json({
      wellId: id,
      narrative: narrative.narrative
    });
  } catch (error) {
    console.error('Error generating narrative:', error);
    res.status(500).json({ error: 'Failed to generate narrative' });
  }
});

export default router;
```

```typescript
// src/components/wells/WellChat.tsx

import { useState } from 'react';
import { useMutation } from '@tanstack/react-query';
import { Button } from '@/components/ui/button';
import { Input } from '@/components/ui/input';
import { Card } from '@/components/ui/card';
import { MessageCircle, Send } from 'lucide-react';
import ReactMarkdown from 'react-markdown';

export function WellChat({ wellId }: { wellId: string }) {
  const [messages, setMessages] = useState<Array<{ role: string; content: string }>>([]);
  const [input, setInput] = useState('');

  const chat = useMutation({
    mutationFn: (message: string) =>
      fetch('/api/chat', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          wellId,
          message,
          conversationHistory: messages
        })
      }).then(r => r.json()),
    onSuccess: (data) => {
      setMessages(prev => [
        ...prev,
        { role: 'user', content: input },
        { role: 'assistant', content: data.response }
      ]);
      setInput('');
    }
  });

  return (
    <Card className="p-6">
      <h3 className="text-lg font-bold mb-4 flex items-center gap-2">
        <MessageCircle className="w-5 h-5" />
        Ask AI About This Well
      </h3>

      <div className="space-y-4 mb-4 max-h-96 overflow-y-auto">
        {messages.length === 0 ? (
          <div className="text-center py-8 text-slate-400">
            <p className="mb-2">Ask me anything about this well!</p>
            <p className="text-sm">Try: "Why is this well undervalued?" or "What are the risks?"</p>
          </div>
        ) : (
          messages.map((msg, idx) => (
            <div
              key={idx}
              className={`p-4 rounded-lg ${
                msg.role === 'user'
                  ? 'bg-emerald-500/10 ml-12'
                  : 'bg-slate-800 mr-12'
              }`}
            >
              <div className="prose prose-sm prose-slate dark:prose-invert">
                <ReactMarkdown>{msg.content}</ReactMarkdown>
              </div>
            </div>
          ))
        )}
      </div>

      <div className="flex gap-2">
        <Input
          placeholder="Ask a question..."
          value={input}
          onChange={(e) => setInput(e.target.value)}
          onKeyDown={(e) => e.key === 'Enter' && !chat.isPending && chat.mutate(input)}
        />
        <Button
          onClick={() => chat.mutate(input)}
          disabled={!input || chat.isPending}
        >
          <Send className="w-4 h-4" />
        </Button>
      </div>
    </Card>
  );
}
```
